cmake_minimum_required(VERSION 3.8)
project(control_docking)
find_package(ament_cmake REQUIRED)
find_package(opennav_docking REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nav2_util REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(angles REQUIRED)
add_library(tag_dock SHARED src/tag_dock.cpp)
target_compile_features(tag_dock PUBLIC cxx_std_17)
ament_target_dependencies(tag_dock opennav_docking pluginlib nav2_util tf2_geometry_msgs angles)
find_library(NON_CHARGING_DOCK_LIBRARY simple_non_charging_dock HINTS /opt/ros/jazzy/lib REQUIRED)
target_link_libraries(tag_dock ${NON_CHARGING_DOCK_LIBRARY})
pluginlib_export_plugin_description_file(opennav_docking_core docks.xml)
install(TARGETS tag_dock LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_dock_tolerance test/test_dock_tolerance.cpp)
  target_include_directories(test_dock_tolerance PRIVATE src)
endif()
ament_package()
