cmake_minimum_required(VERSION 3.8)
project(control_mppi_critic)
find_package(ament_cmake REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(pluginlib REQUIRED)
find_package(interfaces REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(prediction_critic SHARED src/prediction_critic.cpp)
target_compile_features(prediction_critic PUBLIC cxx_std_17)
target_include_directories(prediction_critic PRIVATE include)
ament_target_dependencies(prediction_critic nav2_mppi_controller pluginlib interfaces tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_mppi_controller critics.xml)
install(TARGETS prediction_critic LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_proximity test/test_proximity.cpp)
  target_include_directories(test_proximity PRIVATE include)
endif()
ament_package()
